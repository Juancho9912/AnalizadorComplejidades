function bsearch(A, n, key)
begin
    low <- 1;
    high <- n;
    while (low <= high) do
    begin
        mid <- (low + high) / 2;
        if (A[mid] = key) then
        begin
            return mid;
        end
        else
        begin
            if (A[mid] < key) then
                low <- mid + 1;
            else
                high <- mid - 1;
            end
        end
    end
    return -1;
end
